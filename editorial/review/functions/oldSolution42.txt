function courseSolution41(t,a=CF41.response){const d=t.d;switch(t.type){case'pick':case'inspect':return '<p><b>판단:</b> '+esc(d.options[t.answer])+'</p>';case'multi':return t.answer.map(i=>'<p>✓ '+esc(d.options[i])+'</p>').join('');case'map':return textTable(['자료','연결'],d.rows.map((x,i)=>[x,d.options[t.answer[i]]]));case'sort':return '<p>'+t.answer.map((i,j)=>(j+1)+'. '+esc(d.rows[i])).join('<br>')+'</p>';case'fix':return '<p><b>수정 문장:</b> '+d.segments.map((v,i)=>esc(i===t.answer[0]?d.options[t.answer[1]]:v)).join(' ')+'</p>';case'num':return textTable(['항목','정답'],d.labels.map((v,i)=>[v,f(t.answer[i])]));case'range':return '<p><b>조건을 만족하는 값: '+f(t.answer)+'</b></p>';case'ledger':return textTable(['대안',"편익−명시적 비용",'암묵적 비용','기회비용','순편익'],courseLedger41(t).map(v=>[v.name,v.g,v.implicit,v.oc,v.surplus]));case'build':return '<p>제약을 모두 충족하면 정답입니다. 가능한 예시는 다음과 같습니다.</p>'+textTable(['선택','수량'],d.items.map((v,i)=>[v[0],d.witness[i]]));case'external':return textTable(['활동량',courseExternalLabel41(t)],d.rows.map((v,i)=>[v[0],courseExternal41(t)[i]]));case'public':return textTable(['내 납부','모금 합계','설치',"남은 토큰과 편익의 합"],[0,4,6,10].map(n=>{const z=publicCalc(d.others,n);return[n,z.total,z.lit?'성공':'실패·반환',z.value];}));default:return '';}}
