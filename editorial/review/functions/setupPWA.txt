async function setupPWA(){window.addEventListener('beforeinstallprompt',e=>{e.preventDefault();pendingInstall=e;});window.addEventListener('appinstalled',()=>{pendingInstall=null;toast('앱 설치를 완료했습니다.');});if(location.protocol==='file:'||!window.isSecureContext||!('serviceWorker'in navigator)||!document.querySelector('link[rel="manifest"]'))return;try{swRegistration=await navigator.serviceWorker.register('./sw.js');swRegistration.addEventListener('updatefound',()=>{const w=swRegistration.installing;if(w)w.addEventListener('statechange',()=>{if(w.state==='installed'&&navigator.serviceWorker.controller)toast('새 버전이 준비되었습니다. 설정에서 업데이트를 적용하세요.');});});navigator.serviceWorker.addEventListener('controllerchange',()=>{if(reloadOnController)location.reload();});}catch(err){showWarning('PWA 등록을 완료하지 못했습니다. 일반 웹 학습은 계속할 수 있습니다. '+err.message);}}
