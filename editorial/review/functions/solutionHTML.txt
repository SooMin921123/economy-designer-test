function solutionHTML(s,res=r){const d=s.data;switch(s.type){case'choice':return `<p><b>판단:</b> ${esc(d.options[s.answer])}</p>`;case'multi':return s.answer.map(i=>`<p>✓ ${esc(d.options[i])}</p>`).join('');case'map':return textTable(['자료','연결'],d.rows.map((x,i)=>[x,d.options[s.answer[i]]]));case'numbers':return textTable(['항목','값'],d.labels.map((v,i)=>[v,f(s.answer[i])]));case'ghost':return textTable(['대안',"편익−명시적 비용",'암묵적 비용','기회비용','순편익'],costRows(d.rows).map(x=>[x.name,x.g,x.implicit,x.oc,x.surplus]));case'threshold':return textTable(['B 비용',"B 편익−명시적 비용",'판단'],[[10,26,'B 유일'],[11,25,'B·C 동률'],[12,24,'C 유일']]);case'countermodel':return "<p>C 편익 32.5 → C 편익−명시적 비용 25.5. B 편익−명시적 비용 26은 여전히 최대이며 순편익은 0.5이다.</p>";case'donation':return textTable(['내 납부',"원리금 합계",'설치',"남은 토큰과 편익의 합"],d.choices.map(n=>{const z=publicCalc(d.others,n);return[n,z.total,z.lit?'성공':'실패·반환',z.value];}));case'allocation':return "<p>6+6+6+6=24. 개인별 상한 6과 합계 24를 모두 충족한다.</p>";case'ppf':return '<p>최대 Y=12×(1−18/24)=3개.</p>';case'trade':{const y=num(res?.value),z=tradeCalc(d,Number.isFinite(y)?y:4);return textTable(['항목','현재 제안 또는 예시'],[['X1당 받는Y',f(z.ratio)],['A의 X1 생산 기회비용(Y)',f(z.ca)],['B의 X1 생산 기회비용(Y)',f(z.cb)],['A의 추가 이익(Y)',f(z.gainA)],['B의 추가 이익(Y)',f(z.gainB)]])+`<p>수량 조건을 지키면서 3&lt;Y&lt;6이어야 한다. 예: Y4이면 A 이익 1, B 이익 2.</p>`;}case'write':return '<p>설명 초안은 다시 열어 수정할 수 있습니다. 원고의 근거 확인과 출간 교열은 별도입니다.</p>';default:return '';}}
